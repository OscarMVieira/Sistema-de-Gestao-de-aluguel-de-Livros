USE biblioteca_bd;

-- Elimina a tabela antiga para garantir que não existem erros de estrutura
DROP TABLE IF EXISTS users;

-- Cria a tabela com o campo password preparado para o Hash 
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL, -- Necessário para guardar o hash de 60 caracteres
    documento VARCHAR(20),
    tipoContaId INT DEFAULT 3,
    foto VARCHAR(255) DEFAULT 'default_perfil.png'
);
